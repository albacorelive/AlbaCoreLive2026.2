Chat di testo attiva.
`; currentRemoteUserId = null; currentRemoteUsername = null; currentRemoteFullUid = null; } function cercaAltroUtente(peerIdOverride) { const activePeerId = peerIdOverride || (peer ? peer.id : null); if (!activePeerId) return; nascondiVideoRemoto(); if (currentCall) { currentCall.close(); currentCall = null; } if (currentRoomId) { db.ref('rooms/' + currentRoomId).off(); db.ref('rooms/' + currentRoomId).remove(); currentRoomId = null; } const queueRef = db.ref('queue'); queueRef.once('value', function(snapshot) { const waitingUsers = snapshot.val(); let targetKey = null; let targetItem = null; if (waitingUsers) { const keys = Object.keys(waitingUsers); for (let i = 0; i < keys.length; i++) { if (keys[i] !== userId) { targetKey = keys[i]; targetItem = waitingUsers[targetKey]; break; } } } if (!targetItem) { // Nessuno in attesa: mi metto in coda currentRoomId = 'room_' + userId + '_' + Math.random().toString(36).substring(2, 7); const myQueueRef = db.ref('queue/' + userId); myQueueRef.set({ userId: userId, peerId: activePeerId, roomId: currentRoomId, nickname: nicknameUtente, uid: userUniqueId }); myQueueRef.onDisconnect().remove(); // Ascolto se qualcuno entra nella mia stanza db.ref('rooms/' + currentRoomId).on('value', function(roomSnap) { const roomData = roomSnap.val(); if (!roomData) return; if (roomData.info && roomData.info.peer2 && roomData.info.guestId) { currentRemoteUsername = roomData.info.nick2 || "Utente"; currentRemoteUserId = roomData.info.uid2 || "Sconosciuto"; currentRemoteFullUid = roomData.info.guestId; document.getElementById("remoteLabel").textContent = currentRemoteUsername + " [" + currentRemoteUserId + "]"; const activeAudioTrack = (hasVoice && processedStream) ? processedStream.getAudioTracks()[0] : myStream.getAudioTracks()[0]; const callStream = new MediaStream([activeAudioTrack, myStream.getVideoTracks()[0]]); const call = peer.call(roomData.info.peer2, callStream); currentCall = call; call.on('stream', function(remoteStream) { mostraVideoRemoto(remoteStream); }); call.on('close', function() { nascondiVideoRemoto(); }); } }); } else { // C'è un utente in attesa: mi aggancio alla sua stanza rispettando le regole currentRoomId = targetItem.roomId; const hostPeerId = targetItem.peerId; const hostUserId = targetItem.userId; currentRemoteUsername = targetItem.nickname || "Utente"; currentRemoteUserId = targetItem.uid || "Sconosciuto"; currentRemoteFullUid = hostUserId; document.getElementById("remoteLabel").textContent = currentRemoteUsername + " [" + currentRemoteUserId + "]"; // Rimuovo l'utente dalla coda db.ref('queue/' + hostUserId).remove(); // Scrivo le info nella stanza rispettando la regola .validate di rooms -> info db.ref('rooms/' + currentRoomId + '/info').set({ peer2: activePeerId, guestId: userId, nick2: nicknameUtente, uid2: userUniqueId }); const activeAudioTrack = (hasVoice && processedStream) ? processedStream.getAudioTracks()[0] : myStream.getAudioTracks()[0]; const callStream = new MediaStream([activeAudioTrack, myStream.getVideoTracks()[0]]); const call = peer.call(hostPeerId, callStream); currentCall = call; call.on('stream', function(remoteStream) { mostraVideoRemoto(remoteStream); }); call.on('close', function() { nascondiVideoRemoto(); }); } }); } function cercaAltroUtenteBtn() { if (peer) { cercaAltroUtente(peer.id); } }
